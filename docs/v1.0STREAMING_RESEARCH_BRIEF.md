```python
class SwiftF0:
    # 音频处理常量
    TARGET_SAMPLE_RATE = 16000      # 16kHz采样率
    HOP_LENGTH = 256                # 256采样点 = 16ms帧步长
    FRAME_LENGTH = 1024             # 1024采样点 = 64ms窗口长度
    STFT_PADDING = 384              # 对称填充 (两侧各384采样点)
    MIN_AUDIO_LENGTH = 256          # 最小输入长度

    # 模型频率范围
    MODEL_FMIN = 46.875 Hz          # G1 (最低可检测音高)
    MODEL_FMAX = 2093.75 Hz         # C7 (最高可检测音高)
```

**批处理方法**:
```python
def detect_from_file(audio_path: str) -> PitchResult:
    """
    批处理模式 - 一次性处理整个文件

    输入: 完整音频文件路径
    输出: PitchResult对象包含:
        - pitch_hz: np.ndarray (N帧 x 1)
        - confidence: np.ndarray (N帧 x 1)
        - timestamps: np.ndarray (N帧 x 1)
        - voicing: np.ndarray (N帧 x 1, boolean)

    瓶颈:
    1. 必须读取完整文件 (librosa.load)
    2. ONNX模型期望完整音频输入 (batch_size=1, samples=全长)
    3. 无法处理无限长音频流
    """
    audio, sr = librosa.load(audio_path, sr=None, mono=True)
    # 重采样整个音频
    audio_16k = librosa.resample(audio, orig_sr=sr, target_sr=16000)
    # 一次性推理
    pitch_hz, confidence = self._extract_pitch_and_confidence(audio_16k)
    # 返回完整结果
    return PitchResult(...)
```

**ONNX模型接口**:
```python
def _extract_pitch_and_confidence(audio_16k: np.ndarray) -> Tuple[np.ndarray, np.ndarray]:
    """
    模型输入/输出规格:

    输入:
        - 名称: 'audio' (或动态名称,从session.get_inputs()[0].name获取)
        - 形状: [1, audio_length] (batch=1, 可变长度)
        - 类型: float32
        - 值域: [-1.0, 1.0] (归一化音频波形)

    输出:
        - outputs[0]: pitch_hz, shape=[1, n_frames], float32
        - outputs[1]: confidence, shape=[1, n_frames], float32

    帧数计算:
        n_frames = floor((audio_length - STFT_PADDING*2) / HOP_LENGTH) + 1

    示例:
        输入1秒音频 (16000采样点)
        n_frames = floor((16000 - 768) / 256) + 1 = 60帧
        输出延迟 = 384采样点 / 16000 = 24ms (填充引入)
    """
    ort_inputs = {self.pitch_input_name: audio_16k[None, :].astype(np.float32)}
    outputs = self.pitch_session.run(None, ort_inputs)
    return outputs[0][0], outputs[1][0]
```

**时间戳计算**:
```python
def _calculate_timestamps(n_frames: int) -> np.ndarray:
    """
    精确时间戳计算公式:

    timestamp[i] = (i * HOP_LENGTH + CENTER_OFFSET) / SAMPLE_RATE

    其中:
        CENTER_OFFSET = (FRAME_LENGTH - 1) / 2 - STFT_PADDING
                      = (1024 - 1) / 2 - 384
                      = 127.5 采样点

    物理意义:
        - 每帧代表音频中心位置的时间点
        - 考虑STFT窗口对称性和填充偏移

    示例:
        第0帧: t = (0 * 256 + 127.5) / 16000 = 7.97ms
        第1帧: t = (1 * 256 + 127.5) / 16000 = 23.97ms
        帧间隔: 256 / 16000 = 16ms
    """
    frame_centers = np.arange(n_frames) * self.HOP_LENGTH + self.CENTER_OFFSET
    return frame_centers / self.TARGET_SAMPLE_RATE
```

```python
    min_note_duration: float = 0.05,          # 最小音符时长(秒)

    批处理音符分割算法

    输入: 完整PitchResult (所有帧)
    输出: List[NoteSegment] 每个包含:
        - start: float (开始时间,秒)
        - end: float (结束时间,秒)
        - pitch_median: float (中位音高,Hz)
        - pitch_midi: int (MIDI音符号,0-127)

    算法流程:
    1. 向量化计算MIDI音高: midi = 69 + 12*log2(f/440)
    2. 遍历所有帧,检测音高跳变或语音/非语音切换
    3. 应用静音容忍(grace period)避免过度分割

    5. 合并相邻同音高音符

    瓶颈:
    - 需要完整pitch contour才能计算中位数
    - 需要前瞻(look-ahead)来合并音符
    - 无法实时输出(必须等待整个序列)

    # 关键代码片段
    for i, is_voiced in enumerate(valid_voiced_frames):
        if is_voiced:
            current_median = np.median(current_note_segment["samples"])
            pitch_deviation = abs(midi_pitch - current_median)
            if pitch_deviation >= split_semitone_threshold:
                # 音高跳变 -> 结束当前音符,开始新音符
                notes.append(current_note_segment)
                current_note_segment = new_segment()
        else:
            unvoiced_frames_count += 1
            if unvoiced_duration >= unvoiced_grace_period:
                # 静音超时 -> 结束当前音符
                notes.append(current_note_segment)

    # 后处理: 合并相邻同音高音符
    for current_note in processed_notes[1:]:
        if gap <= frame_period and same_pitch:
```

**调性检测算法**:
```python
def detect_key(notes: List[NoteSegment]) -> Tuple[str, str]:
    """
    Krumhansl-Schmuckler调性检测

    输入: 完整音符序列
    输出: (调名, 调式) 例如 ('C', 'major')

    算法:
    1. 构建音高类直方图 (12个bin: C, C#, D, ...)
       histogram[i] = sum(duration for notes with pitch_class=i)

    2. 归一化直方图
       histogram /= histogram.sum()

    3. 与24个模板相关性分析
       - 12个大调模板 (基于Krumhansl认知实验数据)
       - 12个小调模板
       相关系数 = corrcoef(histogram, rotated_profile)

    4. 返回最高相关系数的调性

    实时挑战:
    - 需要足够统计量(建议>=2秒有声音频)
    - 调性可能变化(需要滑动窗口检测)
    - 初始冷启动问题(无足够数据时降级处理)
    """
    pitch_class_histogram = np.zeros(12)
    for note in notes:
        duration = note.end - note.start
        pitch_class = note.pitch_midi % 12
        pitch_class_histogram[pitch_class] += duration

    # 与大调/小调模板相关性比较
    correlations = []
    for tonic in range(12):
        major_corr = np.corrcoef(histogram, rotate(MAJOR_PROFILE, tonic))[0,1]
        minor_corr = np.corrcoef(histogram, rotate(MINOR_PROFILE, tonic))[0,1]
        correlations.append((KEY_NAMES[tonic], 'major', major_corr))
        correlations.append((KEY_NAMES[tonic], 'minor', minor_corr))

    return max(correlations, key=lambda x: x[2])[:2]
```

## 🎯 实时流式处理需求规格

### 功能性需求

#### FR1: 低延迟音高检测
```
需求: 从麦克风输入到音高输出延迟 < 50ms
当前: ~150ms (批处理模式)
差距: 3倍性能改进

组成:
- 音频采集延迟: ~10ms (硬件/驱动)
- 缓冲延迟: ~16ms (单帧 @ 16kHz)
- 模型推理: <20ms (目标)
- 后处理: <4ms
```

#### FR2: 流式音符分割
```
需求: 实时输出note_on/note_off MIDI事件
挑战:
- 无完整contour,如何计算中位数?
- 如何处理grace period(需要前瞻)?
- 何时确认音符结束?

建议方案:
1. 增量中位数: 使用滑动窗口或running median
2. 延迟输出: note_on立即发送, note_off延迟grace_period确认
3. 状态机: IDLE -> TENTATIVE_START -> ACTIVE_NOTE -> TENTATIVE_END -> IDLE
```

#### FR3: 在线调性检测
```
需求: 动态更新调性判断(支持转调)
方案:
1. 冷启动: 前2-4秒收集统计,使用默认C major
2. 滑动窗口: 每10秒重新检测调性
3. 平滑过渡: 检测到转调时渐进式切换
```

#### FR4: 流式MIDI输出
```
需求: 实时发送MIDI消息而非生成文件
接口:
- USB MIDI设备
- MIDI over BLE (蓝牙)
- 虚拟MIDI端口 (软件内部)
- OSC (Open Sound Control)

消息格式:
- Note On: [0x90, pitch, velocity]
- Note Off: [0x80, pitch, velocity]
- Program Change: [0xC0, instrument_number]
- Pitch Bend: [0xE0, lsb, msb] (用于微分音修正)
```

### 方向2: 实时音频流处理框架

**问题**: 需要高效的音频流缓冲和线程管理

**调研任务**:
```
1. Python音频流库对比
   库名              延迟    跨平台    线程模型        推荐度
   ─────────────────────────────────────────────────
   PyAudio          低      优秀      callback        ⭐⭐⭐⭐⭐
   sounddevice      低      优秀      callback/block  ⭐⭐⭐⭐⭐
   PySoundCard      中      良好      generator       ⭐⭐⭐
   python-soundcard 低      优秀      简洁API         ⭐⭐⭐⭐

2. 流式架构设计模式
   - Callback模式: 音频线程直接调用处理函数 (最低延迟)
   - Queue模式: 生产者-消费者解耦 (更灵活)
   - Ring Buffer模式: 无锁环形缓冲 (高性能)

3. 线程模型
   Option A: 单线程
     Audio Input -> Buffer -> Inference -> MIDI Output
     优点: 简单,无锁
     缺点: 推理阻塞音频采集

   Option B: 双线程
     Thread1: Audio Input -> Ring Buffer
     Thread2: Ring Buffer -> Inference -> MIDI Output
     优点: 解耦,推理不阻塞采集
     缺点: 需要线程同步

   Option C: 三线程 (推荐)
     Thread1: Audio Input -> Buffer1
     Thread2: Buffer1 -> Inference -> Buffer2
     Thread3: Buffer2 -> Note Segmentation -> MIDI Output
     优点: 完全解耦,最佳吞吐
     缺点: 复杂度高
```

**参考架构**:
```python
import sounddevice as sd
import queue
import threading

# 全局缓冲队列
audio_queue = queue.Queue(maxsize=10)
pitch_queue = queue.Queue(maxsize=10)

# 线程1: 音频采集
def audio_callback(indata, frames, time, status):
    """
    由sounddevice实时调用 (音频线程)
    注意: 此回调必须非常快 (<1ms), 否则导致dropout
    """
    if status:
        print(f"Audio status: {status}")
    # 快速放入队列,不做任何处理
    audio_queue.put(indata.copy())

# 线程2: 音高检测
def pitch_detection_worker():
    """
    持续从audio_queue读取,推理,写入pitch_queue
    """
    detector = SwiftF0()
    buffer = np.array([])  # 滑动窗口缓冲

    while True:
        chunk = audio_queue.get()  # 阻塞等待

        # 追加到缓冲
        buffer = np.append(buffer, chunk.flatten())

        # 当缓冲足够时推理
        if len(buffer) >= 1024:
            # 取1024采样点推理
            input_chunk = buffer[:1024]
            buffer = buffer[256:]  # 滑动256采样点 (16ms)

            # ONNX推理
            pitch, confidence = detector._extract_pitch_and_confidence(input_chunk)

            # 输出到下游
            pitch_queue.put({
                'pitch': pitch[0],  # 单帧
                'confidence': confidence[0],
                'timestamp': time.time()
            })

# 线程3: 音符分割和MIDI输出
def note_segmentation_worker():
    """
    维护状态机,实时发送MIDI消息
    """
    state = 'IDLE'
    current_note = None
    grace_counter = 0

    while True:
        frame = pitch_queue.get()

        is_voiced = frame['confidence'] > 0.9
        pitch_midi = hz_to_midi(frame['pitch'])

        if state == 'IDLE':
            if is_voiced:
                # 启动新音符
                current_note = pitch_midi
                send_midi_note_on(pitch_midi, velocity=80)
                state = 'ACTIVE_NOTE'

        elif state == 'ACTIVE_NOTE':
            if not is_voiced:
                # 进入grace period
                grace_counter = 0
                state = 'TENTATIVE_END'
            elif abs(pitch_midi - current_note) > 0.8:
                # 音高跳变 -> 结束旧音符,开始新音符
                send_midi_note_off(current_note)
                current_note = pitch_midi
                send_midi_note_on(pitch_midi, velocity=80)

        elif state == 'TENTATIVE_END':
            if is_voiced and abs(pitch_midi - current_note) < 0.8:
                # 恢复到同一音符
                state = 'ACTIVE_NOTE'
            else:
                grace_counter += 1
                if grace_counter * 0.016 > 0.02:  # 20ms grace period
                    # 确认结束
                    send_midi_note_off(current_note)
                    current_note = None
                    state = 'IDLE'

                    # 如果是音高跳变导致的,立即开始新音符
                    if is_voiced:
                        current_note = pitch_midi
                        send_midi_note_on(pitch_midi, velocity=80)
                        state = 'ACTIVE_NOTE'

# 启动系统
threading.Thread(target=pitch_detection_worker, daemon=True).start()
threading.Thread(target=note_segmentation_worker, daemon=True).start()

# 启动音频流
stream = sd.InputStream(
    channels=1,
    samplerate=16000,
    blocksize=256,  # 16ms块大小
    callback=audio_callback
)

with stream:
    print("实时处理中... 按Ctrl+C停止")
    while True:
```

### 方向3: 增量音符分割算法

**问题**: 当前算法需要完整contour和后向合并

**调研任务**:
```
1. 增量中位数算法
   - P² algorithm (Jain & Chlamtac, 1985)
   - Reservoir sampling median
   - 滑动窗口中位数 (复杂度O(log n)用heap)

2. 状态机设计
   States: IDLE, NOTE_TENTATIVE_START, NOTE_ACTIVE, NOTE_TENTATIVE_END

   Transitions:
   IDLE -> NOTE_TENTATIVE_START: 检测到voiced frame
   NOTE_TENTATIVE_START -> NOTE_ACTIVE: 持续N帧同音高
   NOTE_ACTIVE -> NOTE_TENTATIVE_END: 音高跳变或unvoiced
   NOTE_TENTATIVE_END -> NOTE_ACTIVE: grace period内恢复
   NOTE_TENTATIVE_END -> IDLE: grace period超时

3. 前瞻缓冲设计
   问题: 音符结束需要延迟确认
   方案: 维持grace_period长度的缓冲 (例如20ms = 1-2帧)

   trade-off:
   - 缓冲越大 -> 延迟越高,准确性越好
   - 缓冲越小 -> 延迟越低,可能误切分

   建议: 可配置参数,卡祖笛实时演奏用10ms,录音转MIDI用50ms
```

**伪代码**:
```python
class RealtimeNoteSegmenter:
    def __init__(
        self,
        split_threshold=0.8,
        min_note_frames=3,  # 最小音符帧数 (48ms @ 16ms/frame)
        grace_period_frames=2,  # grace period帧数 (32ms)
    ):
        self.state = 'IDLE'
        self.current_note_pitches = []  # 当前音符的pitch历史
        self.current_note_start = None
        self.grace_counter = 0
        self.note_start_frames = 0

    def process_frame(self, pitch_hz, confidence, timestamp, is_voiced):
        """
        处理单帧,返回MIDI事件列表

        返回格式: [
            {'type': 'note_on', 'pitch': 60, 'time': 1.234},
            {'type': 'note_off', 'pitch': 60, 'time': 1.567},
        ]
        """
        events = []

        if not is_voiced:
            if self.state == 'NOTE_ACTIVE':
                self.state = 'NOTE_TENTATIVE_END'
                self.grace_counter = 0
            elif self.state == 'NOTE_TENTATIVE_END':
                self.grace_counter += 1
                if self.grace_counter >= self.grace_period_frames:
                    # 确认结束
                    if self.note_start_frames >= self.min_note_frames:
                        events.append({
                            'type': 'note_off',
                            'pitch': self._get_median_midi(),
                            'time': timestamp
                        })
                    self._reset_state()
            return events

        # Voiced frame
        midi_pitch = 69 + 12 * np.log2(pitch_hz / 440.0)

        if self.state == 'IDLE':
            # 开始新音符
            self.current_note_start = timestamp
            self.current_note_pitches = [midi_pitch]
            self.note_start_frames = 1
            self.state = 'NOTE_ACTIVE'

            # 立即发送note_on
            events.append({
                'type': 'note_on',
                'pitch': int(round(midi_pitch)),
                'time': timestamp,
                'velocity': 80
            })

        elif self.state == 'NOTE_TENTATIVE_END':
            # 检查是否恢复到同一音符
            median_midi = self._get_median_midi()
            if abs(midi_pitch - median_midi) < self.split_threshold:
                # 恢复
                self.state = 'NOTE_ACTIVE'
                self.current_note_pitches.append(midi_pitch)
                self.note_start_frames += 1
            else:
                # 确认结束并开始新音符
                if self.note_start_frames >= self.min_note_frames:
                    events.append({
                        'type': 'note_off',
                        'pitch': int(round(median_midi)),
                        'time': timestamp
                    })

                # 新音符
                self.current_note_start = timestamp
                self.current_note_pitches = [midi_pitch]
                self.note_start_frames = 1
                self.state = 'NOTE_ACTIVE'
                events.append({
                    'type': 'note_on',
                    'pitch': int(round(midi_pitch)),
                    'time': timestamp,
                    'velocity': 80
                })

        elif self.state == 'NOTE_ACTIVE':
            # 检查音高跳变
            median_midi = self._get_median_midi()
            if abs(midi_pitch - median_midi) >= self.split_threshold:
                # 音高跳变 -> 结束并开始新音符
                if self.note_start_frames >= self.min_note_frames:
                    events.append({
                        'type': 'note_off',
                        'pitch': int(round(median_midi)),
                        'time': timestamp
                    })

                self.current_note_start = timestamp
                self.current_note_pitches = [midi_pitch]
                self.note_start_frames = 1
                events.append({
                    'type': 'note_on',
                    'pitch': int(round(midi_pitch)),
                    'time': timestamp,
                    'velocity': 80
                })
            else:
                # 继续当前音符
                self.current_note_pitches.append(midi_pitch)
                self.note_start_frames += 1
                # 限制历史长度防止内存爆炸
                if len(self.current_note_pitches) > 100:
                    self.current_note_pitches.pop(0)

        return events

    def _get_median_midi(self):
        """增量中位数估计"""
        return np.median(self.current_note_pitches)

    def _reset_state(self):
        self.state = 'IDLE'
        self.current_note_pitches = []
        self.current_note_start = None
        self.grace_counter = 0
        self.note_start_frames = 0
```

# 流式MIDI发送器
class RealtimeMIDISender:
    def __init__(self, port_name=None, instrument=0):
        """
        初始化实时MIDI发送器

        Args:
            port_name: MIDI端口名称,None则创建虚拟端口
            instrument: GM音色号 (0-127)
        """
        if port_name:
            self.port = mido.open_output(port_name)
        else:
            self.port = mido.open_output('SwiftF0 Virtual', virtual=True)

        # 设置音色
        self.send_program_change(instrument)

        # 追踪当前活跃音符(避免重复note_off)
        self.active_notes = set()

    def send_program_change(self, program):
        """改变音色"""
        msg = mido.Message('program_change', program=program)
        self.port.send(msg)
        print(f"音色切换到: {program}")

    def send_note_on(self, pitch, velocity=80):
        """发送note_on消息"""
        pitch = int(round(pitch))
        pitch = max(0, min(127, pitch))  # clamp

        msg = mido.Message('note_on', note=pitch, velocity=velocity)
        self.port.send(msg)
        self.active_notes.add(pitch)
        print(f"Note ON: {pitch} (vel={velocity})")

    def send_note_off(self, pitch):
        """发送note_off消息"""
        pitch = int(round(pitch))
        pitch = max(0, min(127, pitch))

        if pitch in self.active_notes:
            msg = mido.Message('note_off', note=pitch, velocity=0)
            self.port.send(msg)
            self.active_notes.remove(pitch)
            print(f"Note OFF: {pitch}")

    def send_pitch_bend(self, cents):
        """
        发送pitch bend消息(用于微分音修正)

        Args:
            cents: 音分偏移 (-50 ~ +50 cents)
        """
        # MIDI pitch bend 范围: -8192 ~ +8191
        # 默认范围: ±2 semitones = ±200 cents
        # 1 cent = 8192 / 200 = 40.96
        bend_value = int(cents * 40.96)
        bend_value = max(-8192, min(8191, bend_value))

        # 分解为LSB和MSB
        bend_value += 8192  # 转为0~16383
        lsb = bend_value & 0x7F
        msb = (bend_value >> 7) & 0x7F

        msg = mido.Message('pitchwheel', pitch=bend_value - 8192)
        self.port.send(msg)

    def all_notes_off(self):
        """紧急关闭所有音符(避免"stuck notes")"""
        for pitch in list(self.active_notes):
            self.send_note_off(pitch)

        # 发送All Notes Off控制消息
        msg = mido.Message('control_change', control=123, value=0)
        self.port.send(msg)

    def close(self):
        """关闭MIDI端口"""
        self.all_notes_off()
        self.port.close()

# 使用示例
if __name__ == '__main__':
    # 创建MIDI发送器
    midi_sender = RealtimeMIDISender(instrument=56)  # Trumpet

    try:
        # 模拟实时事件流
        events = [
            {'type': 'note_on', 'pitch': 60, 'time': 0.0},
            {'type': 'note_off', 'pitch': 60, 'time': 0.5},
            {'type': 'note_on', 'pitch': 64, 'time': 0.5},
            {'type': 'note_off', 'pitch': 64, 'time': 1.0},
        ]

        start_time = time.time()
        for event in events:
            # 等待到事件时间
            while time.time() - start_time < event['time']:
                time.sleep(0.001)

            # 发送MIDI消息
            if event['type'] == 'note_on':
                midi_sender.send_note_on(event['pitch'])
            elif event['type'] == 'note_off':
                midi_sender.send_note_off(event['pitch'])

        time.sleep(1)  # 等待最后的音符结束

    finally:
        midi_sender.close()

```python
    """
```
